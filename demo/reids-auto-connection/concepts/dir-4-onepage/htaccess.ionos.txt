# Reids Auto Connection - live site on IONOS (uploaded as .htaccess; kept under this
# name in the repo so it never takes effect on the topshelfsolutions.io demo copy).
RewriteEngine On

# One address: https://reidscars.com (host-scoped so the IONOS preview domain still works over http)
RewriteCond %{HTTP_HOST} reidscars\.com$ [NC]
RewriteCond %{HTTPS} off
RewriteRule ^(.*)$ https://reidscars.com/$1 [L,R=301]
RewriteCond %{HTTP_HOST} ^www\.reidscars\.com$ [NC]
RewriteRule ^(.*)$ https://reidscars.com/$1 [L,R=301]

# The old three-page site's addresses
RewriteRule ^application\.htm$ /get-approved.html [L,R=301,NC]
RewriteRule ^Gallery\.htm$ / [L,R=301,NC]
RewriteRule ^index\.htm$ / [L,R=301,NC]

# Speed: compress text, and let browsers keep images/fonts/CSS/JS (pages are always re-checked)
<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/css application/javascript image/svg+xml application/xml text/plain
</IfModule>
<IfModule mod_expires.c>
  ExpiresActive On
  ExpiresByType text/html "access plus 0 seconds"
  ExpiresByType image/webp "access plus 30 days"
  ExpiresByType image/png "access plus 30 days"
  ExpiresByType image/jpeg "access plus 30 days"
  ExpiresByType image/x-icon "access plus 30 days"
  ExpiresByType text/css "access plus 7 days"
  ExpiresByType application/javascript "access plus 7 days"
</IfModule>
